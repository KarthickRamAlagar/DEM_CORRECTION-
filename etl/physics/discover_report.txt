  2 Writes only small count tables to data/processed/sdg/flood_eval/. Valid pixel = observed (flood!=255) AND exclusion==0 AND ref_water!=1

 61 def rd(path, shape):
 62     with rasterio.open(path) as s:
 63         return s.read(1, out_shape=shape, resampling=Resampling.nearest)
